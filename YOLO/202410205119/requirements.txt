# ============================================================================
# 一、核心：目标检测框架与深度学习框架
# ============================================================================

# YOLO 实现（本项目核心）。提供 YOLO 类，负责预训练权重加载、训练、验证、
# 推理、画框、模型导出。>=8.3 才有完整的 YOLOv8 支持与稳定的训练参数。
ultralytics>=8.3.0

# PyTorch 深度学习框架。显式声明，保证与 torchvision 版本成对。
# 从 PyPI 装到的是 CPU 版（形如 2.x.x+cpu），无 NVIDIA 显卡时就用它。
torch>=2.0.0

# torchvision：提供视觉算子与骨干网络实现，主次版本需与 torch 匹配。
torchvision>=0.15.0


# ============================================================================
# 二、图像处理与数值计算
# ============================================================================

# 读写图片、在图上画检测框（predict.py 保存结果图时使用）。
# 注意：cv2.imread / cv2.imwrite 在 Windows 上无法处理含中文的路径，
#       所以本项目统一用 Pillow 读图后再转成 OpenCV 数组。
opencv-python>=4.7.0

# 图像基础操作 + 中文文字渲染（ImageFont / ImageDraw）。
# predict.py 里画「可乐 0.87」这类中文标签依赖它，因为 cv2.putText 只支持 ASCII。
pillow>=10.0.0

# 张量/矩阵运算，torch 与 ultralytics 的底层依赖。
numpy>=1.23.0

# 解析 data.yaml 等配置文件。
PyYAML>=6.0


# ============================================================================
# 三、训练可视化与监控
# ============================================================================

# 绘制 results.png、混淆矩阵、PR 曲线等训练结果图。
matplotlib>=3.7.0

# 打印系统资源占用（内存/显存），Ultralytics 训练日志会用到。
psutil>=5.9.0

# results.csv 表格读写的底层依赖。
polars>=0.20.0

# 训练/推理进度条显示。
tqdm>=4.66.0


# ============================================================================
# 四、Level 5 网页部署（Gradio）
